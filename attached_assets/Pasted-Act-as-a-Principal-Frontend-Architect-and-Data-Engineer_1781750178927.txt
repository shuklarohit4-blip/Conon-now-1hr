MODULE 2: THE CONON BRANDED FALLBACK UI (Error Handling)
If a primary image fails to load or the URL is empty, you must NOT show a broken image icon. You must use a clean, dynamic placeholder using `lucide-react` or `https://placehold.co`.

Implement this strict `<CononImage>` component wrapper for ALL images in the app:
```jsx
import React, { useState } from 'react';
import { PackageSearch } from 'lucide-react'; // Or use placeholder service

const CononImage = ({ src, alt, className }) => {
  const [imgSrc, setImgSrc] = useState(src || '[https://placehold.co/600x400/f1f5f9/475569?text=Conon](https://placehold.co/600x400/f1f5f9/475569?text=Conon)');
  const [hasError, setHasError] = useState(false);

  const fallbackUrl = `https://placehold.co/600x400/f1f5f9/475569?font=montserrat&text=${alt?.replace(/\s+/g, '+') || 'Material'}`;

  return (
    <div className={`relative overflow-hidden ${className}`}>
      {src && !hasError ? (
        <img
          src={imgSrc}
          alt={alt}
          className="w-full h-full object-cover transition-opacity duration-300"
          onError={() => {
            setImgSrc(fallbackUrl);
            setHasError(true);
          }}
        />
      ) : (
        <div className="w-full h-full bg-slate-100 flex flex-col items-center justify-center p-4 border border-slate-200 rounded-xl">
            <PackageSearch className="text-slate-400 mb-1" size={32} />
            <span className="text-[10px] text-slate-500 font-medium text-center line-clamp-2">{alt || 'Material'}</span>
        </div>
      )}
    </div>
  );
};

import React, { useState } from 'react';
import { 
  User, ShieldCheck, Store, HardHat, 
  ArrowRight, Smartphone, Lock, LayoutDashboard,
  Package, Truck, PieChart, LogOut, Bell, 
  Upload, FileText, CheckCircle, Clock, AlertCircle, X
} from 'lucide-react';

const App = () => {
  const [userRole, setUserRole] = useState(null); // 'customer', 'vendor', 'admin'
  const [isLoggedIn, setIsLoggedIn] = useState(false);
  const [onboardingStep, setOnboardingStep] = useState(1);
  const [isVerified, setIsVerified] = useState(false);

  // --- LOGIN / SELECT PORTAL ---
  const LoginView = () => (
    <div className="flex-1 flex flex-col p-8 justify-center animate-in fade-in duration-500">
      <div className="text-center mb-10">
        <h1 className="text-4xl font-black italic text-blue-900 tracking-tighter">CONON</h1>
        <p className="text-[10px] font-black uppercase tracking-[0.3em] text-slate-400">Select Your Portal</p>
      </div>

      <div className="space-y-4">
        {[
          { id: 'customer', title: 'I am a Builder / Owner', icon: <HardHat size={24}/>, desc: 'Order materials in 60 mins', color: 'bg-blue-600' },
          { id: 'vendor', title: 'I am a Supplier / Shop', icon: <Store size={24}/>, desc: 'Manage your store & inventory', color: 'bg-emerald-600' },
          { id: 'admin', title: 'Central Control (Admin)', icon: <ShieldCheck size={24}/>, desc: 'System monitor & analytics', color: 'bg-slate-900' }
        ].map((role) => (
          <button 
            key={role.id}
            onClick={() => { setUserRole(role.id); setIsLoggedIn(true); }}
            className="w-full bg-white border-2 border-slate-100 p-5 rounded-[30px] flex items-center gap-5 hover:border-blue-500 transition-all shadow-sm active:scale-95 text-left"
          >
            <div className={`${role.color} text-white p-3 rounded-2xl shadow-lg`}>{role.icon}</div>
            <div>
              <h3 className="font-black text-slate-800 text-sm">{role.title}</h3>
              <p className="text-[10px] font-bold text-slate-400 uppercase">{role.desc}</p>
            </div>
          </button>
        ))}
      </div>
    </div>
  );

  // --- VENDOR ONBOARDING (Registration & GST Upload) ---
  const VendorOnboarding = () => (
    <div className="flex-1 flex flex-col p-8 animate-in slide-in-from-bottom duration-500">
      <div className="flex justify-between items-center mb-8">
        <button onClick={() => setIsLoggedIn(false)} className="text-slate-400"><X size={20}/></button>
        <div className="flex gap-1">
          <div className={`w-8 h-1 rounded-full ${onboardingStep >= 1 ? 'bg-emerald-500' : 'bg-slate-100'}`}></div>
          <div className={`w-8 h-1 rounded-full ${onboardingStep >= 2 ? 'bg-emerald-500' : 'bg-slate-100'}`}></div>
        </div>
      </div>

      {onboardingStep === 1 ? (
        <>
          <h2 className="text-2xl font-black text-slate-900 mb-2">Register Shop</h2>
          <p className="text-xs text-slate-400 font-bold uppercase mb-8">Basic Business Information</p>
          <div className="space-y-4">
            <input type="text" placeholder="Shop Name (e.g. Sharma Hardware)" className="w-full p-4 rounded-2xl bg-slate-50 border border-slate-100 font-bold text-sm outline-none focus:ring-2 ring-emerald-500/20" />
            <input type="text" placeholder="Owner Name" className="w-full p-4 rounded-2xl bg-slate-50 border border-slate-100 font-bold text-sm outline-none" />
            <input type="tel" placeholder="Mobile Number" className="w-full p-4 rounded-2xl bg-slate-50 border border-slate-100 font-bold text-sm outline-none" />
            <button 
              onClick={() => setOnboardingStep(2)}
              className="w-full bg-emerald-600 text-white py-4 rounded-3xl font-black text-sm uppercase tracking-widest mt-4 shadow-xl shadow-emerald-600/20"
            >
              Next: Documents
            </button>
          </div>
        </>
      ) : (
        <>
          <h2 className="text-2xl font-black text-slate-900 mb-2">Verification</h2>
          <p className="text-xs text-slate-400 font-bold uppercase mb-8">Upload Legal Documents</p>
          <div className="space-y-4">
            <div className="border-2 border-dashed border-slate-200 p-8 rounded-[35px] text-center bg-slate-50 group hover:border-emerald-400 transition cursor-pointer">
              <Upload className="mx-auto text-slate-300 mb-3 group-hover:text-emerald-500" />
              <p className="text-[10px] font-black uppercase text-slate-400">Upload GST Certificate</p>
              <p className="text-[8px] text-slate-300 mt-1 uppercase font-bold">PDF, PNG or JPG</p>
            </div>
            <div className="border-2 border-dashed border-slate-200 p-8 rounded-[35px] text-center bg-slate-50 group hover:border-emerald-400 transition cursor-pointer">
              <FileText className="mx-auto text-slate-300 mb-3 group-hover:text-emerald-500" />
              <p className="text-[10px] font-black uppercase text-slate-400">Shop License / Trade Doc</p>
              <p className="text-[8px] text-slate-300 mt-1 uppercase font-bold">Verification is Mandatory</p>
            </div>
            <button 
              onClick={() => setIsVerified('pending')}
              className="w-full bg-slate-900 text-white py-4 rounded-3xl font-black text-sm uppercase tracking-widest mt-4"
            >
              Submit for Review
            </button>
          </div>
        </>
      )}
    </div>
  );

  // --- VENDOR PENDING STATE ---
  const VendorPending = () => (
    <div className="flex-1 flex flex-col items-center justify-center p-10 text-center animate-in zoom-in duration-300">
      <div className="w-24 h-24 bg-amber-50 text-amber-500 rounded-full flex items-center justify-center mb-6 animate-pulse">
        <Clock size={40} />
      </div>
      <h2 className="text-xl font-black text-slate-800 italic uppercase">Verification Pending</h2>
      <p className="text-[11px] text-slate-400 font-bold mt-4 leading-relaxed uppercase tracking-tighter">
        Our team is reviewing your GST and Trade documents. <br/> This usually takes 24-48 hours.
      </p>
      <div className="mt-8 bg-blue-50 p-4 rounded-2xl flex items-center gap-3 text-left w-full">
         <AlertCircle className="text-blue-500" size={20} />
         <p className="text-[9px] font-bold text-blue-700 uppercase">You will receive an SMS once your shop is active.</p>
      </div>
      <button onClick={() => {setIsVerified(false); setOnboardingStep(1); setIsLoggedIn(false);}} className="mt-10 text-slate-400 text-xs font-black uppercase tracking-widest border-b border-slate-200">Go Back</button>
    </div>
  );

  // --- PLACEHOLDER DASHBOARDS ---
  const CustomerDashboard = () => <div className="p-8 text-center"><h2 className="font-black text-blue-900">CUSTOMER PORTAL ACTIVE</h2><button onClick={()=>setIsLoggedIn(false)} className="mt-4 bg-slate-100 px-4 py-2 rounded-xl text-[10px] font-black uppercase">Logout</button></div>;
  const AdminDashboard = () => <div className="p-8 text-center font-black">ADMIN CONTROL<br/><button onClick={()=>setIsLoggedIn(false)} className="mt-4 bg-slate-900 text-white px-4 py-2 rounded-xl text-[10px]">Logout</button></div>;

  return (
    <div className="flex items-center justify-center min-h-screen bg-slate-200 p-2 font-sans">
      <div className="w-full max-w-[420px] h-[880px] bg-white rounded-[60px] shadow-2xl overflow-hidden relative border-[14px] border-slate-900 flex flex-col">
        {!isLoggedIn && <LoginView />}
        {isLoggedIn && userRole === 'customer' && <CustomerDashboard />}
        {isLoggedIn && userRole === 'admin' && <AdminDashboard />}
        {isLoggedIn && userRole === 'vendor' && !isVerified && <VendorOnboarding />}
        {isLoggedIn && userRole === 'vendor' && isVerified === 'pending' && <VendorPending />}
      </div>
    </div>
  );
};

export default App;
